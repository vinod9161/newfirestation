CREATE TABLE services (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    code VARCHAR(50) UNIQUE,
    description TEXT NULL,
    is_active TINYINT(1) DEFAULT 1,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    INDEX idx_services_active (is_active)
) ENGINE=InnoDB;

CREATE TABLE pricing_rules (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    service_id BIGINT UNSIGNED NOT NULL,

    -- CONDITIONS
    min_sq_ft INT NULL,
    max_sq_ft INT NULL,

    min_height DECIMAL(8,2) NULL,
    max_height DECIMAL(8,2) NULL,

    min_gathering INT NULL,
    max_gathering INT NULL,

    min_hours INT NULL,
    max_hours INT NULL,

    -- PRICING
    rate DECIMAL(10,2) NOT NULL,
    rate_type ENUM('fixed','per_sq_ft','per_person','per_hour') NOT NULL,

    -- CONTROL
    priority INT DEFAULT 1,
    is_active TINYINT(1) DEFAULT 1,

    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    -- FOREIGN KEY
    CONSTRAINT fk_pricing_service
        FOREIGN KEY (service_id)
        REFERENCES services(id)
        ON DELETE CASCADE,

    -- INDEXES (IMPORTANT)
    INDEX idx_pricing_service (service_id),
    INDEX idx_pricing_active (is_active),
    INDEX idx_pricing_priority (priority),
    INDEX idx_pricing_service_active (service_id, is_active)
) ENGINE=InnoDB;


CREATE TABLE payments (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    service_id BIGINT UNSIGNED NOT NULL,

    amount DECIMAL(10,2) NOT NULL,

    status ENUM('pending','success','failed') DEFAULT 'pending',

    transaction_id VARCHAR(150) NULL,
    order_id VARCHAR(150) NULL,

    payment_gateway VARCHAR(50) NULL,

    response JSON NULL,

    paid_at TIMESTAMP NULL,

    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    -- FOREIGN KEY
    CONSTRAINT fk_payment_service
        FOREIGN KEY (service_id)
        REFERENCES services(id)
        ON DELETE CASCADE,

    -- INDEXES
    INDEX idx_payment_service (service_id),
    INDEX idx_payment_status (status),
    INDEX idx_payment_txn (transaction_id)
) ENGINE=InnoDB;
